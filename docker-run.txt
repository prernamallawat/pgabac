# pg_abac — How to Run
# All commands below are run from the pgabac/ directory unless stated otherwise.

# ═══════════════════════════════════════════════════════
# STEP 1 — Start the database (first time)
# ═══════════════════════════════════════════════════════
# This builds the Docker image, compiles the extension,
# and loads the full University database automatically.
# It will take 1-2 minutes the first time.

docker compose up --build

# Wait until you see this line in the logs:
#   database system is ready to accept connections
# Then open a NEW terminal window and continue to Step 2.

# ═══════════════════════════════════════════════════════
# STEP 2 — Connect to the database
# ═══════════════════════════════════════════════════════
# Run this in a new terminal window while docker compose
# is still running in the background.

docker exec -it pg-abac psql -U postgres -d university_abac

# ═══════════════════════════════════════════════════════
# STEP 3 — Verify everything loaded correctly
# ═══════════════════════════════════════════════════════
# Run this inside the psql session from Step 2.

\i /pg_abac/abac/verify_abac.sql

# ═══════════════════════════════════════════════════════
# OPTIONAL — Run demo and benchmarks
# ═══════════════════════════════════════════════════════
# Also run from inside the psql session.

\i /pg_abac/demo/demo.sql

# Benchmarks — run these in the container shell (not psql).
# To open a container shell:  docker exec -it pg-abac bash

pgbench -U postgres -d university_abac -f benchmark/baseline.sql -T 30
pgbench -U postgres -d university_abac -f benchmark/abac_enabled.sql -T 30
pgbench -U postgres -d university_abac -f benchmark/policy_complexity.sql -T 30

# ═══════════════════════════════════════════════════════
# STOPPING AND RESTARTING
# ═══════════════════════════════════════════════════════

# Stop the container (data is preserved)
docker compose down

# Start again without rebuilding (data is preserved)
docker compose up

# Full clean reset — deletes all data and rebuilds from scratch
docker compose down -v
docker compose up --build
