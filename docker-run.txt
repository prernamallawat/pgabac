# Docker run instructions for final ABAC project

# 1. From the unzipped project folder on your Mac:
docker run --name pg-abac -e POSTGRES_PASSWORD=postgres -e POSTGRES_DB=university_abac -p 5434:5432 -v "$PWD":/pg_abac -d postgres:15

# 2. Enter container:
docker exec -it pg-abac bash

# 3. Install build tools inside container:
apt update
apt install -y build-essential postgresql-server-dev-15

# 4. Build and install extension:
cd /pg_abac
make clean
make PG_CONFIG=/usr/lib/postgresql/15/bin/pg_config
make install PG_CONFIG=/usr/lib/postgresql/15/bin/pg_config

# 5. Load complete project:
psql -U postgres -d university_abac -f load_initial_project.sql

# 6. Run demo:
psql -U postgres -d university_abac -f demo/demo_script.sql

# 7. Run verification:
psql -U postgres -d university_abac -f abac/verify_abac.sql

# 8. Run benchmarks:
pgbench -U postgres -d university_abac -f benchmark/baseline.sql -T 30
pgbench -U postgres -d university_abac -f benchmark/abac_enabled.sql -T 30
pgbench -U postgres -d university_abac -f benchmark/policy_complexity.sql -T 30

# docker stop pg-abac
